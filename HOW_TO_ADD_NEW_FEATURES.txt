Step 1: Add Database Table

CREATE TABLE testimonials (
  id INT PRIMARY KEY AUTO_INCREMENT,
  client_name VARCHAR(100) NOT NULL,
  client_position VARCHAR(100),
  client_company VARCHAR(100),
  testimonial TEXT NOT NULL,
  rating INT DEFAULT 5,
  client_image VARCHAR(255),
  is_featured BOOLEAN DEFAULT FALSE,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

Step 2: Add Backend Route
File: backend/routes/testimonials.js

const express = require('express');
const router = express.Router();
const db = require('../config/db');
const verifyToken = require('../middleware/auth');

router.get('/', async (req, res) => {
  try {
    const [rows] = await db.query('SELECT * FROM testimonials ORDER BY created_at DESC');
    res.json(rows);
  } catch (error) {
    res.status(500).json({ message: 'Server error', error: error.message });
  }
});

router.post('/', verifyToken, async (req, res) => {
  const { client_name, client_position, client_company, testimonial, rating } = req.body;
  try {
    const [result] = await db.query(
      'INSERT INTO testimonials (client_name, client_position, client_company, testimonial, rating) VALUES (?, ?, ?, ?, ?)',
      [client_name, client_position, client_company, testimonial, rating]
    );
    res.status(201).json({ message: 'Testimonial added', id: result.insertId });
  } catch (error) {
    res.status(500).json({ message: 'Server error', error: error.message });
  }
});

module.exports = router;

Step 3: Connect Route in server.js

Step 4: Add Frontend Page
File: frontend/src/pages/Testimonials.js

import React, { useState, useEffect } from 'react';
import { FaStar } from 'react-icons/fa';
import api from '../config/api';
import './Testimonials.css';

function Testimonials() {
  const [testimonials, setTestimonials] = useState([]);

  useEffect(() => {
    fetchTestimonials();
  }, []);

  const fetchTestimonials = async () => {
    try {
      const response = await api.get('/testimonials');
      setTestimonials(response.data);
    } catch (error) {
      console.error('Error:', error);
    }
  };

  const renderStars = (rating) => {
    return [...Array(5)].map((_, i) => (
      <FaStar key={i} color={i < rating ? '#ffd700' : '#ddd'} />
    ));
  };

  return (
    <div className="testimonials">
      <div className="container">
        <h1 className="page-title">Client Testimonials</h1>
        <div className="testimonials-grid">
          {testimonials.map((item) => (
            <div key={item.id} className="testimonial-card">
              <div className="rating">{renderStars(item.rating)}</div>
              <p className="testimonial-text">"{item.testimonial}"</p>
              <div className="client-info">
                <strong>{item.client_name}</strong>
                <span>{item.client_position} at {item.client_company}</span>
              </div>
            </div>
          ))}
        </div>
      </div>
    </div>
  );
}

export default Testimonials;

FEATURE 3: DARK MODE
Toggle between light and dark themes!

Step 1: Add Context
File: frontend/src/contexts/ThemeContext.js

import React, { createContext, useState, useEffect } from 'react';

export const ThemeContext = createContext();

export function ThemeProvider({ children }) {
  const [darkMode, setDarkMode] = useState(false);

  useEffect(() => {
    const saved = localStorage.getItem('darkMode');
    if (saved) {
      setDarkMode(saved === 'true');
    }
  }, []);

  const toggleDarkMode = () => {
    setDarkMode(!darkMode);
    localStorage.setItem('darkMode', !darkMode);
  };

  return (
    <ThemeContext.Provider value={{ darkMode, toggleDarkMode }}>
      <div className={darkMode ? 'dark-mode' : 'light-mode'}>
        {children}
      </div>
    </ThemeContext.Provider>
  );
}

Step 2: Add Dark Mode Styles
File: frontend/src/index.css

.dark-mode {
  --bg-primary: #1a1a1a;
  --bg-secondary: #2d2d2d;
  --text-primary: #ffffff;

}

  --bg-primary: #ffffff;
  --bg-secondary: #f8f9fa;

Step 3: Add Toggle Button
File: frontend/src/components/DarkModeToggle.js

import React, { useContext } from 'react';
import { FaMoon, FaSun } from 'react-icons/fa';
import { ThemeContext } from '../contexts/ThemeContext';

function DarkModeToggle() {
  const { darkMode, toggleDarkMode } = useContext(ThemeContext);

  return (
    <button onClick={toggleDarkMode} className="dark-mode-toggle">
      {darkMode ? <FaSun /> : <FaMoon />}
    </button>
  );
}

export default DarkModeToggle;

Step 4: Wrap App with Provider
In frontend/src/index.js:

import { ThemeProvider } from './contexts/ThemeContext';

  <ThemeProvider>
    <App />
  </ThemeProvider>
);
